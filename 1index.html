async function investigate() { const q = document.getElementById("query").value.trim(); if (!q) return; currentSearchQuery = q; document.getElementById("output").innerHTML = `
Searching indexed database for ${esc(q)}...
`; try { const res = await fetch("/api/investigate?q=" + encodeURIComponent(q)); const data = await res.json(); console.log("API Response Data:", data); // Add this line render(data); } catch (err) { console.error("Search error:", err); document.getElementById("output").innerHTML = `
Error executing search query.
`; } }
